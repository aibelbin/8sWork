# Find Jobs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A shared job board (`/jobs`). People post jobs to it with a "Share this job" checkbox on the application form or a share-only form, and anyone signed in can "Track this" a posting into their own applications.

**Architecture:**
- A new `postings` table that every signed-in user can read and only the creator can change (enforced by RLS).
- A nullable `applications.posting_id` link.
- Two pure mapping functions, `postingFromApplication` and `applicationFromPosting`.
- A small `syncShare` helper that the application actions call after saving.
- The board is a server-rendered page with two small client buttons.

**Tech Stack:** Next.js 16 App Router (Cache Components: per-user data inside `<Suspense>`), Supabase (supabase-js, RLS), zod 4, Vitest, Tailwind/shadcn (Base UI).

**Spec:** `docs/superpowers/specs/2026-11-14-find-jobs-design.md`

## Global Constraints

- Postings are anonymous: the UI never shows `created_by`.
- Never copy status, rounds, resume_version, source or application notes into a posting.
- "Today" is the IST date from `istDate()` in `src/lib/dates.ts`, computed on the server.
- Every `(app)` page keeps its per-user data inside `<Suspense fallback={<PageLoading />}>`.
- No new dependencies.
- Typecheck with `npm run typecheck`, test with `npm test`. No local `next build`, since the Mac is fanless.
- Commit messages carry no AI attribution.

```sql
create table public.postings (
  id uuid primary key default gen_random_uuid(),
  created_by uuid default auth.uid() references public.profiles (id) on delete cascade,
  origin public.posting_origin not null default 'shared',
  company text not null check (char_length(company) between 1 and 200),
  role text not null check (char_length(role) between 1 and 200),
  apply_method public.apply_method not null,
  job_url text,
  contact_email text,
  job_type public.job_type,
  location text,
  salary text,
  deadline date,
  details text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create index postings_created_at_idx on public.postings (created_at desc);
create trigger postings_set_updated_at before update on public.postings
  for each row execute function public.set_updated_at();

alter table public.postings enable row level security;
-- The board is shared: every signed-in user reads every posting; only the creator changes it.
create policy "postings: read all" on public.postings for select to authenticated using (true);
create policy "postings: insert own" on public.postings for insert to authenticated
  with check (created_by = (select auth.uid()));
create policy "postings: update own" on public.postings for update to authenticated
  using (created_by = (select auth.uid())) with check (created_by = (select auth.uid()));
create policy "postings: delete own" on public.postings for delete to authenticated
  using (created_by = (select auth.uid()));
grant select, insert, update, delete on public.postings to authenticated;

alter table public.applications
  add column posting_id uuid references public.postings (id) on delete set null;
create index applications_posting_id_idx on public.applications (posting_id);
```

```ts
import type { ISODate } from './dates';
import type { Application, Posting } from './domain';
import { suggestFollowUp } from './follow-up';

export type PostingFields = Pick<
  Posting,
  'company' | 'role' | 'apply_method' | 'job_url' | 'contact_email' | 'job_type' | 'location' | 'salary' | 'deadline'
>;
export type NewApplication = Omit<Application, 'id' | 'user_id' | 'created_at' | 'updated_at'>;

// Only these fields go on the shared board; status, rounds, resume, source and notes stay private.
export function postingFromApplication(app: PostingFields): PostingFields {
  const { company, role, apply_method, job_url, contact_email, job_type, location, salary, deadline } = app;
  return { company, role, apply_method, job_url, contact_email, job_type, location, salary, deadline };
}

export function applicationFromPosting(posting: Posting, today: ISODate): NewApplication {
  return {
    ...postingFromApplication(posting),
    status: 'to_apply',
    source: 'friend',
    source_detail: 'Find jobs',
    resume_version: null,
    applied_on: null,
    follow_up_on: suggestFollowUp({ status: 'to_apply', deadline: posting.deadline, applied_on: null, rounds: [] }, 'created', today),
    notes: posting.details,
    posting_id: posting.id,
  };
}
```

- [ ] **Step 1: Create `src/lib/share-db.ts`**

```ts
import type { Application } from './domain';
import { postingFromApplication } from './postings';
import type { ServerSupabase } from './supabase/server';

// Keeps the application's own posting in step with the "Share this job" checkbox.
// An application tracking someone else's posting is left alone: that job is already on the board.
export async function syncShare(supabase: ServerSupabase, userId: string, app: Application, share: boolean): Promise<string | null> {
  let ownsPosting = false;
  if (app.posting_id) {
    const { data } = await supabase.from('postings').select('created_by').eq('id', app.posting_id).maybeSingle();
    if (data && data.created_by !== userId) return null;
    ownsPosting = !!data;
  }
  if (share && ownsPosting) {
    const { error } = await supabase.from('postings').update(postingFromApplication(app)).eq('id', app.posting_id!);
    return error?.message ?? null;
  }
  if (share) {
    const { data, error } = await supabase.from('postings').insert(postingFromApplication(app)).select('id').single();
    if (error) return error.message;
    const { error: linkError } = await supabase.from('applications').update({ posting_id: data.id }).eq('id', app.id);
    return linkError?.message ?? null;
  }
  if (ownsPosting) {
    const { error } = await supabase.from('postings').delete().eq('id', app.posting_id!);
    return error?.message ?? null;
  }
  return null;
}
```

```ts
export async function createApplication(_prev: FormState, formData: FormData): Promise<FormState> {
  const { supabase, user } = await requireUser();
```

```ts
'use server';

import { revalidatePath } from 'next/cache';
import { redirect } from 'next/navigation';
import { z } from 'zod';
import { istDate } from '@/lib/dates';
import type { Posting } from '@/lib/domain';
import type { ActionResult } from '@/lib/form-state';
import { applicationFromPosting } from '@/lib/postings';
import { requireUser } from '@/lib/supabase/server';

export async function trackPosting(id: string): Promise<ActionResult> {
  if (!z.uuid().safeParse(id).success) return { error: 'Invalid job' };
  const { supabase } = await requireUser();
  const { data: posting } = await supabase.from('postings').select('*').eq('id', id).maybeSingle();
  if (!posting) return { error: 'That job was removed' };
  const { data: existing } = await supabase.from('applications').select('id').eq('posting_id', id).maybeSingle();
  if (existing) redirect(`/applications/${existing.id}`);
  const { data, error } = await supabase
    .from('applications')
    .insert(applicationFromPosting(posting as Posting, istDate()))
    .select('id')
    .single();
  if (error) return { error: error.message };
  revalidatePath('/', 'layout');
  redirect(`/applications/${data.id}`);
}

export async function removePosting(id: string): Promise<ActionResult> {
  if (!z.uuid().safeParse(id).success) return { error: 'Invalid job' };
  const { supabase } = await requireUser();
  const { error } = await supabase.from('postings').delete().eq('id', id);
  if (error) return { error: error.message };
  revalidatePath('/jobs');
  return {};
}
```

```tsx
    .from('postings')
    .select('*')
    .or(`deadline.is.null,deadline.gte.${today}`)
    .order('created_at', { ascending: false });
  if ((JOB_TYPES as readonly string[]).includes(type)) query = query.eq('job_type', type);
  const [{ data, error }, { data: tracked }] = await Promise.all([
    query,
    supabase.from('applications').select('id, posting_id').not('posting_id', 'is', null),
  ]);
  if (error) throw new Error(error.message);

  const trackedBy = new Map((tracked ?? []).map((a) => [a.posting_id as string, a.id as string]));
  // Search runs in JS so user text never goes into a PostgREST filter string.
  const needle = q.trim().toLowerCase();
  const postings = (data as Posting[]).filter(
    (p) => !needle || p.company.toLowerCase().includes(needle) || p.role.toLowerCase().includes(needle),
  );

  return (
    <>
      <form role="search" className="flex flex-wrap gap-2">
        <Input name="q" aria-label="Search" placeholder="Search company or role" defaultValue={q} className="w-full sm:w-56" />
        <NativeSelect name="type" aria-label="Filter by job type" options={JOB_TYPES} placeholder="All job types" defaultValue={type} className="w-auto" />
        <Button type="submit" variant="outline">Apply</Button>
      </form>

      {postings.length === 0 ? (
        <p className="text-muted-foreground">
          No jobs shared yet. <Link href="/jobs/new" className="underline">Share a job</Link>
        </p>
      ) : (
        <ul className="grid gap-3">
          {postings.map((p) => {
            const appId = trackedBy.get(p.id);
            const mine = p.created_by === user.id;
            const facts = [p.salary, p.location, p.job_type && label(p.job_type), p.deadline && `Apply by ${formatDate(p.deadline)}`].filter(Boolean);
            return (
              <li key={p.id} className="space-y-3 rounded-lg border p-4">
                <div className="flex flex-wrap items-start justify-between gap-2">
                  <div>
                    <p className="font-medium">{p.company}</p>
                    <p className="text-sm text-muted-foreground">{p.role}</p>
                  </div>
                  <div className="flex items-center gap-2">
                    {appId ? (
                      <Link href={`/applications/${appId}`}><Badge variant="secondary">Tracking</Badge></Link>
                    ) : (
                      !mine && <TrackButton id={p.id} />
                    )}
                    {mine && <RemovePostingButton id={p.id} />}
                  </div>
                </div>
                {facts.length > 0 && <p className="text-sm">{facts.join(' · ')}</p>}
                <p className="flex flex-wrap gap-3 text-sm">
                  <span className="text-muted-foreground">Apply via {label(p.apply_method).toLowerCase()}</span>
                  {p.job_url && <a href={p.job_url} target="_blank" rel="noreferrer" className="underline">Open posting</a>}
                  {p.contact_email && <a href={`mailto:${p.contact_email}`} className="underline">{p.contact_email}</a>}
                </p>
                {p.details && <p className="text-sm whitespace-pre-line text-muted-foreground">{p.details}</p>}
              </li>
            );
          })}
        </ul>
      )}
    </>
  );
}

export default function JobsPage({ searchParams }: { searchParams: SearchParams }) {
  return (
    <div className="space-y-6">
      <div className="flex flex-wrap items-center justify-between gap-3">
```

```ts
export async function sharePosting(_prev: FormState, formData: FormData): Promise<FormState> {
  const { supabase } = await requireUser();
  const values = Object.fromEntries(formData) as Record<string, string>;
  const result = postingSchema.safeParse(values);
  if (!result.success) return { errors: z.flattenError(result.error).fieldErrors, values };
  const { error } = await supabase.from('postings').insert(result.data);
  if (error) return { message: error.message, values };
  revalidatePath('/jobs');
  redirect('/jobs');
}
```
